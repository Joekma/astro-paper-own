## 概述

高性能网络编程是构建高并发服务器的关键技术，涉及 IO 模型、线程池、异步编程等多个方面。

## IO 模型

### 阻塞 IO（BIO）

```python
import socket

def blocking_server(port):
    server = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    server.bind(('0.0.0.0', port))
    server.listen(128)

    while True:
        client, addr = server.accept()
        data = client.recv(1024)
        client.sendall(data)
        client.close()
```

| 特点 | 说明 |
|------|------|
| 阻塞等待 | accept/recv 都会阻塞 |
| 资源占用 | 每个连接一个线程 |
| 适用场景 | 连接数较少的场景 |

### 非阻塞 IO（NIO）

```python
import socket
import select

def nonblocking_server(port):
    server = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    server.setblocking(False)
    server.bind(('0.0.0.0', port))
    server.listen(128)

    clients = []

    while True:
        rlist, _, _ = select.select([server] + clients, [], [])

        for sock in rlist:
            if sock is server:
                client, addr = server.accept()
                client.setblocking(False)
                clients.append(client)
            else:
                try:
                    data = sock.recv(1024)
                    if data:
                        sock.sendall(data)
                    else:
                        sock.close()
                        clients.remove(sock)
                except:
                    sock.close()
                    clients.remove(sock)
```

### IO 多路复用

#### select

```python
import select

def select_server(port):
    server = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    server.bind(('0.0.0.0', port))
    server.listen(128)

    inputs = [server]
    outputs = []

    while inputs:
        readable, writable, exceptional = select.select(inputs, outputs, inputs)

        for s in readable:
            if s is server:
                client, _ = s.accept()
                inputs.append(client)
            else:
                data = s.recv(1024)
                if data:
                    outputs.append(s)
                else:
                    s.close()
                    inputs.remove(s)

        for s in writable:
            s.sendall(b'ACK')
            outputs.remove(s)
            inputs.remove(s)
            s.close()
```

#### poll

```python
import select

def poll_server(port):
    server = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    server.bind(('0.0.0.0', port))
    server.listen(128)

    poll = select.poll()
    poll.register(server, select.POLLIN)

    fd_to_socket = {server.fileno(): server}

    while True:
        events = poll.poll()

        for fd, event in events:
            sock = fd_to_socket[fd]

            if sock is server:
                client, _ = sock.accept()
                poll.register(client, select.POLLIN)
                fd_to_socket[client.fileno()] = client
            else:
                data = sock.recv(1024)
                if data:
                    poll.modify(sock, select.POLLOUT)
                else:
                    poll.unregister(sock)
                    sock.close()
                    del fd_to_socket[fd]
```

#### epoll（Linux）

```python
import select

def epoll_server(port):
    server = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    server.bind(('0.0.0.0', port))
    server.listen(128)
    server.setblocking(False)

    epoll = select.epoll()
    epoll.register(server.fileno(), select.EPOLLIN)

    fd_to_socket = {server.fileno(): server}

    while True:
        events = epoll.poll()

        for fd, event in events:
            sock = fd_to_socket[fd]

            if sock is server:
                client, _ = sock.accept()
                client.setblocking(False)
                epoll.register(client.fileno(), select.EPOLLIN)
                fd_to_socket[client.fileno()] = client
            elif event & select.EPOLLIN:
                data = sock.recv(1024)
                if data:
                    epoll.modify(fd, select.EPOLLOUT)
                else:
                    epoll.unregister(fd)
                    sock.close()
                    del fd_to_socket[fd]
            elif event & select.EPOLLOUT:
                sock.sendall(b'ACK')
                epoll.modify(fd, select.EPOLLIN)
```

### 异步 IO

#### asyncio

```python
import asyncio

async def handle_client(reader, writer):
    data = await reader.read(1024)
    writer.write(data)
    await writer.drain()
    writer.close()

async def async_server(host, port):
    server = await asyncio.start_server(
        handle_client, host, port
    )
    async with server:
        await server.serve_forever()

asyncio.run(async_server('0.0.0.0', 8888))
```

```python
from aiohttp import web

async def handle(request):

    return web.Response(text=data)

app = web.Application()
app.router.add_post('/api', handle)

web.run_app(app, host='0.0.0.0', port=8080)
```

### 生产者-消费者模式

```python
import threading
import queue

def producer(q):
    for i in range(100):
        q.put(i)

def consumer(q):
    while True:
        item = q.get()
        if item is None:
            break
        print(f'处理: {item}')

q = queue.Queue()
threads = []

for _ in range(3):
    t = threading.Thread(target=consumer, args=(q,))
    t.start()
    threads.append(t)

producer(q)

for _ in range(3):
    q.put(None)

for t in threads:
    t.join()
```

### 进程池

```python
from concurrent.futures import ProcessPoolExecutor

def cpu_intensive_task(n):
    return sum(i * i for i in range(n))

with ProcessPoolExecutor(max_workers=4) as executor:
    results = list(executor.map(cpu_intensive_task, [1000000] * 10))
```

## 连接池

### 数据库连接池

```python
import queue
import pymysql

class ConnectionPool:
    def __init__(self, max_connections=10, **kwargs):
        self.pool = queue.Queue(max_connections)
        self.kwargs = kwargs

        for _ in range(max_connections):
            conn = pymysql.connect(**kwargs)
            self.pool.put(conn)

    def get_connection(self, timeout=None):
        return self.pool.get(timeout=timeout)

    def return_connection(self, conn):
        self.pool.put(conn)

    def close_all(self):
        while not self.pool.empty():
            conn = self.pool.get()
            conn.close()
```

## 性能优化

### 连接复用

```python
import socket

class ReusableSocket:
    def __init__(self, host, port):
        self.host = host
        self.port = port
        self.sock = None

    def connect(self):
        if not self.sock:
            self.sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
            self.sock.connect((self.host, self.port))
            self.sock.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)

    def send(self, data):
        self.sock.sendall(data)
        return self.sock.recv(1024)

    def close(self):
        if self.sock:
            self.sock.close()
            self.sock = None
```

### Nagle 算法

```python
sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)

# 禁用 Nagle（小数据高实时场景）
sock.setsockopt(socket.IPPROTO_TCP, socket.TCP_NODELAY, 1)

# 启用 Nagle（大数据高吞吐场景）
sock.setsockopt(socket.IPPROTO_TCP, socket.TCP_NODELAY, 0)
```

### 缓冲区优化

```python
# 设置缓冲区大小
sock.setsockopt(socket.SOL_SOCKET, socket.SO_RCVBUF, 1024 * 1024)
sock.setsockopt(socket.SOL_SOCKET, socket.SO_SNDBUF, 1024 * 1024)

# 禁用 Nagle

```

## 小结

高性能网络编程要点：

- **IO 模型选择**：epoll > poll > select > 阻塞 IO
- **线程池**：复用线程，减少创建开销
- **异步编程**：asyncio 协程处理高并发
- **连接池**：复用连接，减少连接建立开销
- **缓冲区优化**：合理设置缓冲区大小
