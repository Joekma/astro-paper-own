## 概述

海量数据处理是后端面试的核心主题，本篇文章总结常见面试题及解决方案。

## 核心思想

### 分而治之

将大问题拆分为小问题，分别处理后合并结果：

```
海量数据 ──▶ 分片处理 ──▶ 结果合并
     │            │            │
     ▼            ▼            ▼
  分割文件      Map 阶段      Reduce 阶段
```

### 外排序

数据量超过内存时，使用外部排序：

```
1. 分段读取数据到内存
2. 每段内部排序
3. 归并各段有序数据
```

### Hash 映射

通过 Hash 将数据分布到不同机器或文件：

```python
def get_partition(key, n):
    return hash(key) % n
```

## Top K 问题

### 问题描述

在大规模数据中找出出现频率最高的 K 个元素。

### 堆排序解法

```python
import heapq
from collections import Counter

def top_k_frequent(nums, k):
    # 统计词频
    counter = Counter(nums)

    # 使用小顶堆，保持 k 个最大
    return heapq.nlargest(k, counter.keys(), key=counter.get)

# 示例
nums = [1, 1, 1, 2, 2, 3, 3, 3, 3]
print(top_k_frequent(nums, 2))  # [3, 1]
```

### 分治 + 堆

```python
class TopK:
    def __init__(self, k):
        self.k = k
        self.heap = []

    def add(self, num):
        if len(self.heap) < self.k:
            heapq.heappush(self.heap, (self.frequency(num), num))
        elif num > self.heap[0][1]:
            heapq.heapreplace(self.heap, (self.frequency(num), num))

    def result(self):
        return [item[1] for item in self.heap]
```

## 海量数据去重

### 问题描述

在数十亿条数据中找出不重复的元素。

### Bitmap 方案

```python
class BloomFilter:
    def __init__(self, size):
        self.size = size
        self.bits = [0] * size

    def add(self, value):
        positions = self._get_positions(value)
        for pos in positions:
            self.bits[pos] = 1

    def contains(self, value):
        positions = self._get_positions(value)
        return all(self.bits[pos] == 1 for pos in positions)

    def _get_positions(self, value):
        h1 = hash(value) % self.size
        h2 = hash(str(value) + 'x') % self.size
        return [h1, h2]
```

### Hash 分桶

```python
def deduplicate_huge_file(filename, bucket_count=100):
    buckets = [set() for _ in range(bucket_count)]

    # 分桶
    with open(filename, 'r') as f:
        for line in f:
            key = line.strip()
            bucket_id = hash(key) % bucket_count
            buckets[bucket_id].add(key)

    # 各桶内部去重
    results = []
    for i, bucket in enumerate(buckets):
        with open(f'bucket_{i}.txt', 'w') as f:
            for item in bucket:
                f.write(item + '\n')
        results.extend(bucket)

    return results
```

## 海量数据合并

### 多个有序文件合并

```python
import heapq

def merge_sorted_files(file_paths, output_path):
    with open(output_path, 'w') as out:
        # 初始化堆，每文件取首行
        heap = []
        file_handles = []

        for i, path in enumerate(file_paths):
            f = open(path, 'r')
            file_handles.append(f)
            line = f.readline()
            if line:
                heapq.heappush(heap, (line, i))

        # 归并
        while heap:
            line, i = heapq.heappop(heap)
            out.write(line)

            next_line = file_handles[i].readline()
            if next_line:
                heapq.heappush(heap, (next_line, i))

        # 关闭文件
        for f in file_handles:
            f.close()
```

## 分布统计问题

### 找出中位数

```python
def find_median_large_file(filename):
    min_heap = []
    max_heap = []

    with open(filename, 'r') as f:
        for num in f:
            num = int(num.strip())

            # 维持两个堆的平衡
            if not max_heap or num <= -max_heap[0]:
                heapq.heappush(max_heap, -num)
            else:
                heapq.heappush(min_heap, num)

            # 平衡两个堆
            if len(max_heap) > len(min_heap) + 1:
                heapq.heappush(min_heap, -heapq.heappop(max_heap))
            elif len(min_heap) > len(max_heap):
                heapq.heappush(max_heap, -heapq.heappop(min_heap))

    # 获取中位数
    if len(max_heap) > len(min_heap):
        return -max_heap[0]
    else:
        return (-max_heap[0] + min_heap[0]) / 2
```

### 统计独立用户数

```python
def count_unique_users_large_file(filename):
    user_ids = set()
    seen_before = set()

    with open(filename, 'r') as f:
        for line in f:
            user_id = line.strip()

            if user_id in seen_before:
                continue

            if len(user_ids) < 1000000:
                user_ids.add(user_id)
            else:
                seen_before.update(user_ids)
                user_ids = set()

    return len(seen_before) + len(user_ids)
```

## 海量 URL 处理

### URL 去重

```python
class URLDeduper:
    def __init__(self, bucket_count=1000):
        self.buckets = [set() for _ in range(bucket_count)]
        self.seen = set()

    def add(self, url):
        if url in self.seen:
            return False

        bucket_id = hash(url) % self.bucket_count
        if len(self.buckets[bucket_id]) < 1000000:
            self.buckets[bucket_id].add(url)
            self.seen.add(url)
            return True

        # 桶满，转移到已见集合
        self.seen.update(self.buckets[bucket_id])
        self.buckets[bucket_id] = set()
        self.buckets[bucket_id].add(url)
        self.seen.add(url)
        return True
```

## 查询问题

### 快速查询

```python
class LargeDataIndex:
    def __init__(self):
        self.index = {}  # key -> file positions

    def build(self, filename):
        with open(filename, 'r') as f:
            for pos, line in enumerate(f):
                key = self.extract_key(line)
                if key not in self.index:
                    self.index[key] = []
                self.index[key].append(pos)

    def query(self, key):
        positions = self.index.get(key, [])
        results = []
        with open(self.filename, 'r') as f:
            for pos in positions:
                f.seek(pos)
                results.append(f.readline())
        return results
```

## MapReduce 思想

```python
# 模拟 MapReduce
def map_reduce(input_file, mapper, reducer):
    # Map 阶段
    mapped = []
    with open(input_file, 'r') as f:
        for line in f:
            mapped.extend(mapper(line))

    # Shuffle 阶段
    shuffled = {}
    for key, value in mapped:
        if key not in shuffled:
            shuffled[key] = []
        shuffled[key].append(value)

    # Reduce 阶段
    results = {}
    for key, values in shuffled.items():
        results[key] = reducer(key, values)

    return results

# 示例：统计词频
def word_count_mapper(line):
    return [(word, 1) for word in line.split()]

def word_count_reducer(word, counts):
    return sum(counts)

result = map_reduce('data.txt', word_count_mapper, word_count_reducer)
```

## 常见面试题

### Q1: 如何在 10 亿个整数中找出前 1000 个最大的？

```python
def top_1000_in_billion(filename):
    heap = []

    with open(filename, 'r') as f:
        for i, num in enumerate(f):
            num = int(num.strip())
            if len(heap) < 1000:
                heapq.heappush(heap, num)
            elif num > heap[0]:
                heapq.heapreplace(heap, num)

    return sorted(heap, reverse=True)
```

### Q2: 如何找出两个大文件的共同记录？

```python
def find_common_records(file1, file2):
    # 使用布隆过滤器
    bloom = BloomFilter(size=100000000)

    # 加载文件1到布隆过滤器
    with open(file1, 'r') as f:
        for line in f:
            bloom.add(line.strip())

    # 检查文件2
    result = []
    with open(file2, 'r') as f:
        for line in f:
            if bloom.contains(line.strip()):
                result.append(line.strip())

    return result
```

### Q3: 如何在 100G 数据中找到中位数？

```python
def find_median_100g(filename, chunk_size=1000000):
    # 分桶统计
    bucket_counts = [0] * 10000

    with open(filename, 'r') as f:
        chunk = []
        for line in f:
            num = int(line.strip())
            chunk.append(num)

            if len(chunk) >= chunk_size:
                for n in chunk:
                    bucket_counts[n // 10000] += 1
                chunk = []

    # 找到中位数所在的桶
    total = sum(bucket_counts)
    target = total // 2

    cumulative = 0
    bucket_id = 0
    for i, count in enumerate(bucket_counts):
        cumulative += count
        if cumulative >= target:
            bucket_id = i
            break

    # 桶内排序找中位数
    bucket_min = bucket_id * 10000
    bucket_max = (bucket_id + 1) * 10000

    bucket_numbers = []
    with open(filename, 'r') as f:
        for line in f:
            num = int(line.strip())
            if bucket_min <= num < bucket_max:
                bucket_numbers.append(num)

    bucket_numbers.sort()
    return bucket_numbers[len(bucket_numbers) // 2]
```

## 小结

海量数据处理核心技巧：

- **分而治之**：大问题拆小问题
- **哈希分桶**：均匀分布，缩小问题规模
- **堆结构**：快速找出 Top K
- **外部排序**：内存不足时的解决方案
- **位图/布隆过滤器**：高效去重和判断存在
